104inline float worley2D(float x, float y, uint64_t seed = 0) noexcept {
105 int xi = (int)floor(x);
106 int yi = (int)floor(y);
107 float minDist = 9999.0f;
108
109 // On parcourt les cellules voisines pour trouver le point le plus proche
110 for (int j = -1; j <= 1; ++j) {
111 for (int i = -1; i <= 1; ++i) {
112 int cx = xi + i;
113 int cy = yi + j;
114
115 // point aléatoire dans la cellule
116 float fx = random2D(cx, cy, seed);
117 float fy = random2D(cx, cy, seed + 1337);
118
119 // position absolue du point
120 float px = cx + fx;
121 float py = cy + fy;
122
123 float dx = px - x;
124 float dy = py - y;
125 float dist = sqrt(dx*dx + dy*dy);
126
127 if (dist < minDist)
128 minDist = dist;
129 }
130 }

132 return minDist; // souvent dans [0,1] après normalisation
133}

154inline float grad(int hash, float x, float y) {
155 int h = hash & 7; // 8 directions
156 float u = h < 4 ? x : y;
157 float v = h < 4 ? y : x;
158 return ((h & 1) ? -u : u) + ((h & 2) ? -2.0f*v : 2.0f*v);

161inline float simplex2D(float xin, float yin, uint64_t seed = 0) {
162 static uint8_t perm[512];
163 static bool initialized=false;
164 if(!initialized){ initPerm(perm, seed); initialized=true; }
165
166 const float F2 = 0.5f*(sqrt(3.0f)-1.0f);
167 float s = (xin + yin) * F2;
168 int i = floor(xin + s);
169 int j = floor(yin + s);
170 const float G2 = (3.0f - sqrt(3.0f))/6.0f;
171 float t = (i + j) * G2;
172 float X0 = i - t, Y0 = j - t;
173 float x0 = xin - X0, y0 = yin - Y0;
174
175 int i1 = (x0 > y0) ? 1 : 0;
176 int j1 = (x0 > y0) ? 0 : 1;
177
178 float x1 = x0 - i1 + G2;
179 float y1 = y0 - j1 + G2;
180 float x2 = x0 - 1.0f + 2.0f * G2;
181 float y2 = y0 - 1.0f + 2.0f * G2;
182
183 int ii = i & 255;
184 int jj = j & 255;
185
186 float n0,n1,n2;
187
188 float t0 = 0.5f - x0*x0 - y0*y0;
189 n0 = (t0<0)?0:pow(t0,4) * grad(perm[ii+perm[jj]], x0, y0);
190
191 float t1 = 0.5f - x1*x1 - y1*y1;
192 n1 = (t1<0)?0:pow(t1,4) * grad(perm[ii+i1+perm[jj+j1]], x1, y1);
193
194 float t2 = 0.5f - x2*x2 - y2*y2;
195 n2 = (t2<0)?0:pow(t2,4) * grad(perm[ii+1+perm[jj+1]], x2, y2);
196
197 return (70.0f * (n0 + n1 + n2)) * 0.5 + 0.5; // valeur brute [0,1] approximative
198}
