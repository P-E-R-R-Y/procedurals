40inline uint64_t mix64(uint64_t x) noexcept {
41 // Advance with GOLDEN_64 (safe, no FP math)
42 x += GOLDEN_64;
43 // Scramble safely with XOR and multiply by GOLDEN_64 again
44 x ^= (x >> 30);
45 x *= GOLDEN_64; // multiplication is modulo 2^64, safe
46 x ^= (x >> 27);
47 x *= GOLDEN_64;
48 x ^= (x >> 31);
49 return x;
50}
